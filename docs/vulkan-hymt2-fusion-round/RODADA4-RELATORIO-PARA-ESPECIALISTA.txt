RMS_NORM + MUL (linha 16431 do arquivo). Removi o gate inútil do seletor de
pipeline, para que exista um único ponto de decisão.

                               const struct ggml_cgraph * cgraph,
                               int node_idx, std::initializer_list<enum ggml_op> ops) {
      if (!ggml_can_fuse(cgraph, node_idx, ops)) {
          return false;
      }

      if (ops.size() == 2 && ops.begin()[0] == GGML_OP_RMS_NORM && ops.begin()[1] == GGML_OP_MUL) {

          // conservadorismo: so o Adreno 840 foi testado.
          if (ctx->device->disable_fused_rms_norm_mul) {
              return false;
          }
          // ... resto das restrições específicas da fusão ...
      }
